root = Path('H:/git/Multi-Workers')

mapping_text = design.split('## §8 AC → VC 映射表\n', 1)[1].split('## §9 ', 1)[0]
mapping = re.findall(r'^\| (AC-\d{3}) \| (.*?) \| (VC-\d{3}) \| (.*?) \|$', mapping_text, re.M)
assert [item[0] for item in mapping] == expected_ac
assert [item[2] for item in mapping] == expected_vc
assert all(item[0] == vc[4] and item[2] == vc[0] for item, vc in zip(mapping, blocks))
sections = re.findall(r'^## §(\d+) ', design, re.M)

assert len(re.findall(r'^### D-\d{3} ', design, re.M)) == 12
assert len(re.findall(r'^### F-\d{2} ', design, re.M)) == 11
assert len(re.findall(r'^\| C\d{2} \|', design, re.M)) == 16
mermaids = re.findall(r'^```mermaid\n(.*?)^```', design, re.M | re.S)
assert len(mermaids) == 12
for graph in mermaids:
    assert graph.startswith('graph TD\n')
    ids = set(re.findall(r'^\s*([A-Za-z0-9_]+)(?:\[|\{|\s+-->)', graph, re.M))
    ids.update(re.findall(r'\|\s*([A-Za-z0-9_]+)(?:\[|\{|\s*$)', graph, re.M))
    assert len(ids) <= 15, (len(ids), graph)
    assert '-->' not in re.sub(r'-->\|[^"|]+\|', '', graph), 'quoted or missing edge label'
assert 'draft-for-review' in design and '未获用户批准' in design
assert '## §11 精确候选写面与批准条件' in design
all_ac = sorted(set(re.findall(r'AC-\d{3}', spec)))
