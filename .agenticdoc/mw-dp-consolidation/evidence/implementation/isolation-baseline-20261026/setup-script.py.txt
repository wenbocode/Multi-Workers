def sha(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def fresh(path: Path, data: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as handle:
        handle.write(data)
    assert path.read_bytes() == data


def dump(path: Path, value: object) -> None:
    fresh(path, (json.dumps(value, ensure_ascii=False, indent=2) + '\n').encode('utf-8'))


def linked(path: Path) -> bool:
    value = path.lstat()
    return path.is_symlink() or bool(getattr(value, 'st_file_attributes', 0) & 0x400)


def run_git(cwd: Path, *args: str) -> bytes:
    result = subprocess.run(['git', '-c', 'core.quotepath=false', *args], cwd=cwd, capture_output=True)
    assert result.returncode == 0, (args, result.returncode, result.stderr.decode('utf-8', 'replace'))
    return result.stdout

# Structural invariants of the remap, never a product verification claim.
old_tasks = {task['id']: task for task in source_dag['tasks']}
by_id = {task['id']: task for task in new_dag['tasks']}
ancestors: dict[str, set[str]] = {}
def visit(task_id: str, stack: set[str]) -> set[str]:
    assert task_id not in stack, ('cycle', task_id)
    if task_id not in ancestors:
        result: set[str] = set()
        for dependency in by_id[task_id]['deps']:
            result.add(dependency)
            result.update(visit(dependency, stack | {task_id}))
        ancestors[task_id] = result
    return ancestors[task_id]
for task in new_dag['tasks']:

    for field in ('deps', 'stage', 'ac_refs', 'vc_refs', 'goal', 'requirements', 'type', 'execution'):
        assert task[field] == original[field], (task['id'], field)
    visit(task['id'], set())
