def add(n, slug, stage, role, deps, refs, files, goal, req, extra_tests=(), direct=False):
    paths = [P + f for f in files.get('py', [])] + [T + f for f in files.get('ts', [])] + [U + f for f in files.get('test_ts', [])] + [F + f for f in files.get('framework', [])]
    task_key = f'dpc-t{n:02}-{slug}'
    evidence = f'.agenticdoc/mw-dp-consolidation/evidence/implementation/{task_key}'
    if role in ('research', 'review'):
        evidence += '.md'
    tests = [p for p in paths if '/test_' in p or '/test/suite/' in p] + list(extra_tests)
    commands = []
    py_tests = [p[len(P):] for p in tests if p.startswith(P)]
    ts_tests = [p[len('packages/coding-agent/'):] for p in tests if p.startswith(U)]
    frame_tests = [p for p in tests if p.startswith(F)]
    if py_tests:
        commands.append({'cwd': 'packages/multi-workers', 'argv': ['python', '-m', 'pytest', '-p', 'no:cacheprovider', '-q'] + py_tests})
    if ts_tests:
        commands.append({'cwd': 'packages/coding-agent', 'argv': ['node', '../../node_modules/vitest/dist/cli.js', '--run'] + ts_tests})
    if frame_tests:
        commands.append({'cwd': '.', 'argv': ['python', '-m', 'pytest', '-p', 'no:cacheprovider', '-q'] + frame_tests})
    tasks.append({'id': f'T{n:02}', 'task_key': task_key, 'card': f'tasks/T{n:02}-{slug}.md', 'stage': stage,
                  'type': role, 'execution': 'pm-direct' if direct else 'worker', 'deps': [f'T{i:02}' for i in deps],
                  'ac_refs': [f'AC-{i:03}' for i in refs], 'vc_refs': [f'VC-{i:03}' for i in refs],
                  'product_write_face': paths, 'evidence': evidence, 'commands': commands,
                  'goal': goal, 'requirements': req, 'status': 'not-queued', 'code_status': '未开始' if role == 'coding' and not direct else '不适用', 'verification_status': '未验证'})
